(function(){var MS=window.MeetingSimRoom,el=document.getElementById('meeting-sim-modal'),t=document.getElementById('meeting-sim-tab-during');if(!MS)throw new Error('no API');console.assert(MS.PHASES.join(',')==='pre,during,post','PHASES');MS.open(document.getElementById('btn-open-meeting-sim'));console.assert(el.style.display==='flex','open');MS.setPhase('during',t);console.assert(MS.getPhase()==='during','phase');console.assert(MS.buildHTML('during').indexOf('线上会议室')!==-1,'html');MS.close();console.log('MeetingSim console assertions ok');})();
